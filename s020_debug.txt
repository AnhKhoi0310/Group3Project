"""Collect and normalize reviews for one analysis target."""

from collections.abc import Callable
from dataclasses import dataclass

from app.adapters.review_source import ReviewSourceError
from app.services.normalizer import (
    ReviewNormalizationError,
    normalize_reviews_with_skips,
)


@dataclass(frozen=True)
class IngestionOutcome:
    reviews: list[dict]
    reviews_found: int
    reviews_skipped: int


class IngestionError(Exception):
    """Ingestion failed. The category is safe to record and report."""

    def __init__(self, category: str) -> None:
        super().__init__(category)
        self.category = category


def collect_reviews(
    target_id: int,
    source_url: str,
    fetch: Callable[[str], list[dict]],
) -> IngestionOutcome:
    """Fetch from the target's stored URL and return only usable reviews."""
    if not source_url or not source_url.strip():
        raise IngestionError("invalid_source")

    try:
        raw_reviews = fetch(source_url)
    except ReviewSourceError as exc:
        raise IngestionError(exc.category) from exc
    except Exception as exc:
        raise IngestionError("unexpected") from exc

    try:
        reviews, skipped = normalize_reviews_with_skips(
            raw_reviews,
            str(target_id),
        )
    except ReviewNormalizationError as exc:
        raise IngestionError("unexpected") from exc

    return IngestionOutcome(
        reviews=reviews,
        reviews_found=len(reviews),
        reviews_skipped=skipped,
    )

def _target_not_found() -> ApiError:
    return ApiError(
        status.HTTP_404_NOT_FOUND,
        "target_not_found",
        "We couldn't find that target. It may not exist or you may not have access.",
    )


def _record_failure(
    session: Session,
    attempt: IngestionAttempt,
    category: str,
) -> None:
    try:
        fail_attempt(session, attempt, category)
    except SQLAlchemyError:
        session.rollback()
        logger.exception(
            "Could not record failure for ingestion attempt %s",
            attempt.id,
        )


@router.post("/{target_id}/ingestions", response_model=IngestionRead)
def start_ingestion(
    target_id: int,
    user: Annotated[User, Depends(get_current_user)],
    session: Annotated[Session, Depends(get_session)],
    fetch: Annotated[Callable[[str], list[dict]], Depends(get_review_fetcher)],
) -> IngestionRead:
    """Collect, normalize, and persist reviews for an owned target."""

    # Verify ownership before creating an attempt or contacting the provider.
    try:
        target = get_target_for_owner(session, target_id, owner_id=user.id)
    except SQLAlchemyError:
        logger.exception("Could not load target %s for user %s", target_id, user.id)
        raise ApiError(
            status.HTTP_500_INTERNAL_SERVER_ERROR,
            "unexpected",
            "Something went wrong on our side. Please try again.",
        ) from None

    if target is None:
        raise _target_not_found()

    # Record the attempt before contacting the external provider.
    try:
        attempt = create_attempt(session, target.id)
    except SQLAlchemyError:
        session.rollback()
        logger.exception("Could not create ingestion attempt for target %s", target.id)
        raise ApiError(
            status.HTTP_500_INTERNAL_SERVER_ERROR,
            "unexpected",
            "Something went wrong on our side. Please try again.",
        ) from None

    # Always use the source URL stored on the owned target.
    try:
        outcome = collect_reviews(target.id, target.source_url, fetch)
    except IngestionError as exc:
        logger.warning(
            "Ingestion failed for target %s (%s)",
            target.id,
            exc.category,
            exc_info=True,
        )

        _record_failure(session, attempt, exc.category)

        status_code, message = INGESTION_ERRORS.get(
            exc.category,
            INGESTION_ERRORS["unexpected"],
        )

        raise ApiError(
            status_code,
            f"ingestion_{exc.category}",
            message,
        ) from None

    # Save reviews and mark the attempt successful.
    try:
        save_reviews(
            session,
            target_id=target.id,
            source_platform=target.platform,
            reviews=outcome.reviews,
        )

        completed = complete_attempt(
            session,
            attempt,
            reviews_found=outcome.reviews_found,
            reviews_skipped=outcome.reviews_skipped,
        )

    except SQLAlchemyError:
        session.rollback()
        logger.exception("Could not persist ingestion for target %s", target.id)

        # Record the failed outcome after rolling back the review transaction.
        try:
            failed_attempt = session.get(IngestionAttempt, attempt.id)

            if failed_attempt is not None:
                fail_attempt(
                    session,
                    failed_attempt,
                    "persistence_error",
                )

        except SQLAlchemyError:
            session.rollback()
            logger.exception(
                "Could not record persistence failure for attempt %s",
                attempt.id,
            )

        raise ApiError(
            status.HTTP_500_INTERNAL_SERVER_ERROR,
            "ingestion_persistence_error",
            "We collected the reviews but couldn't save them. Please try again.",
        ) from None

    return IngestionRead(
        attempt_id=completed.id,
        target_id=target.id,
        status=completed.status,
        reviews_found=completed.reviews_found,
        reviews_skipped=completed.reviews_skipped,
    )

            "unexpected",
            "Something went wrong on our side. Please try again.",
        ) from None

"""Normalize raw Google Maps reviews into a consistent format."""

import math


class ReviewNormalizationError(ValueError):
    """A raw review contains invalid data."""


def normalize_review(raw_review: dict, place_id: str) -> dict:
    """Convert one SerpApi review into the canonical review structure."""
    if not isinstance(raw_review, dict):
        raise ReviewNormalizationError("Review must be a dictionary.")

    if not isinstance(place_id, str) or not place_id.strip():
        raise ReviewNormalizationError("A business identifier is required.")

    rating = raw_review.get("rating")

    if rating is not None:
        if isinstance(rating, bool) or not isinstance(rating, (int, float)):
            raise ReviewNormalizationError("Review rating must be numeric.")

        if not math.isfinite(rating) or not 1 <= rating <= 5:
            raise ReviewNormalizationError("Review rating must be between 1 and 5.")

        rating = float(rating)

    user = raw_review.get("user")
    if not isinstance(user, dict):
        user = {}

    author = user.get("name")
    author = author.strip() or None if isinstance(author, str) else None

    text = raw_review.get("snippet")
    if not isinstance(text, str) or not text.strip():
        extracted = raw_review.get("extracted_snippet")
        text = extracted.get("original") if isinstance(extracted, dict) else None

    text = text.strip() or None if isinstance(text, str) else None

    date = raw_review.get("date")
    date = date.strip() or None if isinstance(date, str) else None

    review_id = raw_review.get("review_id")
    review_id = review_id.strip() or None if isinstance(review_id, str) else None

    return {
        "review_id": review_id,
        "source": "google_maps",
        "place_id": place_id.strip(),
        "author": author,
        "rating": rating,
        "text": text,
        "date": date,
    }


def normalize_reviews_with_skips(
    raw_reviews: list,
    place_id: str,
) -> tuple[list[dict], int]:
    """Normalize a batch, skipping malformed, incomplete, and duplicate records.

    A usable canonical review must have non-empty text and a rating from 1 to 5.
    The returned integer counts every skipped record.
    """
    if not isinstance(raw_reviews, list):
        raise ReviewNormalizationError("Reviews must be a list.")

    if not isinstance(place_id, str) or not place_id.strip():
        raise ReviewNormalizationError("A business identifier is required.")

    normalized_reviews = []
    seen_review_ids = set()
    skipped = 0

    for raw_review in raw_reviews:
        try:
            normalized = normalize_review(raw_review, place_id)
        except ReviewNormalizationError:
            skipped += 1
            continue

        if not normalized["text"] or normalized["rating"] is None:
            skipped += 1
            continue

        review_id = normalized["review_id"]
        if review_id is not None:
            if review_id in seen_review_ids:
                skipped += 1
                continue
            seen_review_ids.add(review_id)

        normalized_reviews.append(normalized)

    return normalized_reviews, skipped


def normalize_reviews(raw_reviews: list, place_id: str) -> list[dict]:
    """Compatibility wrapper returning only usable normalized reviews."""
    normalized, _skipped = normalize_reviews_with_skips(raw_reviews, place_id)
    return normalized

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models import Review


def save_reviews(
    session: Session,
    target_id: int,
    source_platform: str,
    reviews: list[dict],
) -> int:
    """Save usable reviews, avoiding re-inserting known source review IDs.

    Ownership is inherited from target_id. Callers must verify that the
    authenticated user owns the target before calling this repository.
    """
    source_ids = {
        review["review_id"] for review in reviews if review.get("review_id") is not None
    }

    existing_ids: set[str] = set()
    if source_ids:
        query = select(Review.source_review_id).where(
            Review.target_id == target_id,
            Review.source_review_id.in_(source_ids),
        )
        existing_ids = set(session.scalars(query).all())

    saved = 0
    for item in reviews:
        source_id = item.get("review_id")
        if source_id is not None and source_id in existing_ids:
            continue

        session.add(
            Review(
                target_id=target_id,
                source_review_id=source_id,
                source_platform=source_platform,
                reviewer_name=item.get("author"),
                rating=item["rating"],
                text=item["text"],
                review_date=item.get("date"),
            )
        )
        if source_id is not None:
            existing_ids.add(source_id)
        saved += 1

    return saved


def list_reviews_for_target(session: Session, target_id: int) -> list[Review]:
    """Return reviews belonging to one already-authorized target."""
    query = (
        select(Review)
        .where(Review.target_id == target_id)
        .order_by(Review.created_at.desc(), Review.id.desc())
    )
    return list(session.scalars(query).all())
